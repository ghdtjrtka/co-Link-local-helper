cmake_minimum_required(VERSION 3.20)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../VERSION.json" COLINK_VERSIONS)
string(JSON COLINK_HELPER_VERSION GET "${COLINK_VERSIONS}" helper)
project(colink_inspect_radio VERSION ${COLINK_HELPER_VERSION} LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(inspect-radio.rc.in ${CMAKE_CURRENT_BINARY_DIR}/inspect-radio.rc @ONLY)
add_executable(inspect-radio inspect-radio.cpp ${CMAKE_CURRENT_BINARY_DIR}/inspect-radio.rc)
target_compile_features(inspect-radio PRIVATE cxx_std_17)
target_link_libraries(inspect-radio PRIVATE setupapi)
if(MSVC)
    target_compile_options(inspect-radio PRIVATE /guard:cf)
    target_link_options(inspect-radio PRIVATE /guard:cf /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT)
endif()
